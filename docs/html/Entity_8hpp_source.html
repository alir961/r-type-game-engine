166 bool hit(const std::shared_ptr<Entity> &entity) {
167 const auto& [thisX, thisY] = getComponent<PositionComponent>()->getPositions();
168 const auto& [thisWidth, thisHeight] = getComponent<SizeComponent>()->getSize();
169 const auto& [thisRectWidth, thisRectHeight] = getComponent<SizeRectComponent>()->getSize();
170 float thisActualWidth = thisWidth * thisRectWidth;
171 float thisActualHeight = thisHeight * thisRectHeight;
172 const auto& [entityX, entityY] = entity->getComponent<PositionComponent>()->getPositions();
173 const auto& [entityWidth, entityHeight] = entity->getComponent<SizeComponent>()->getSize();
174 const auto& [entityRectWidth, entityRectHeight] = entity->getComponent<SizeRectComponent>()->getSize();
175 float entityActualWidth = entityWidth * entityRectWidth;
176 float entityActualHeight = entityHeight * entityRectHeight;
177
178 if (this == entity.get()) {
179 return false;
180 }
181 return (thisX < entityX + entityActualWidth &&
182 thisX + thisActualWidth > entityX &&
183 thisY < entityY + entityActualHeight &&
184 thisY + thisActualHeight > entityY);
185 }
