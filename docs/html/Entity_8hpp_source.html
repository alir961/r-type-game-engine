223 bool hit(const std::shared_ptr<Entity> &entity) {
224 const auto& [thisX, thisY] = getComponent<PositionComponent>()->getPosition();
225 const auto& [thisWidth, thisHeight] = getComponent<SizeComponent>()->getSize();
226 const auto& [thisRectWidth, thisRectHeight] = getComponent<SizeRectComponent>()->getSize();
227 float thisActualWidth = thisWidth * thisRectWidth;
228 float thisActualHeight = thisHeight * thisRectHeight;
229 const auto& [entityX, entityY] = entity->getComponent<PositionComponent>()->getPosition();
230 const auto& [entityWidth, entityHeight] = entity->getComponent<SizeComponent>()->getSize();
231 const auto& [entityRectWidth, entityRectHeight] = entity->getComponent<SizeRectComponent>()->getSize();
232 float entityActualWidth = entityWidth * entityRectWidth;
233 float entityActualHeight = entityHeight * entityRectHeight;
234
235 if (this == entity.get()) {
236 return false;
237 }
238 return (thisX < entityX + entityActualWidth &&
239 thisX + thisActualWidth > entityX &&
240 thisY < entityY + entityActualHeight &&
241 thisY + thisActualHeight > entityY);
242 }
